import express from "express";
import cors from "cors";
import mysql from "mysql2/promise";
import path from "path";
import multer from "multer";
import fs from "fs/promises";

const app = express();
// Registering Middleware
app.use(cors());

// Registering Middleware
app.use(express.json());

const connection = await mysql.createConnection(
    {
        host: "localhost",
        user: "root",
        password: "03895590",
        database: "liu"
    }
);

// Register uploads folder as a static file
app.use(
    "/uploads",
    express.static(path.join(import.meta.dirname, "uploads"))
);

const storage = multer.diskStorage({
    destination: (req, file, cb) => {
        cb(null, path.join(import.meta.dirname, "uploads"));
    },
    filename: (req, file, cb) => {
        const uniqueName = Date.now() + "-" + file.originalname;
        cb(null, uniqueName);
    }
});

const upload = multer( { storage } );

// Registering End Point
app.get(
    "/majors",
    async (req, res) => {
        const [rows] = await connection.execute("SELECT MajorCode, Description FROM major");
        res.json(rows);
    }
);

// Registering End Point
app.get(
    "/students/:searchText",
    async (req, res) => {
        
        let sql = "SELECT s.*, m.Description MajorDescription " +
            "FROM students s INNER JOIN major m on s.Major = m.MajorCode";

        let searchText = req.params.searchText;

        if (searchText != 0) {
            searchText = searchText.toLowerCase();
            sql += " WHERE LOWER(Fname) LIKE '%" + searchText + "%' ";
            sql += "    OR LOWER(Lname) LIKE '%" + searchText + "%' ";
            sql += "    OR LOWER(Email) LIKE '%" + searchText + "%' ";
        }

        console.log(sql);
        const [rows] = await connection.execute(sql);

        res.json(rows);
    }
);

// Registering End Point
app.post(
    "/students",
    upload.single("Portrait"),
    async (req, res) => {
        console.log(req.body);
        
        let sql = "INSERT INTO students (StdID, Fname, Lname, Email, Major, Address, Profile) ";
        sql += " VALUES (?, ?, ?, ?, ?, ?, ?)";

        console.log(sql);

        const result = await connection.execute(
            sql,
            [
                req.body.StdID,
                req.body.Fname,
                req.body.Lname,
                req.body.Email,
                req.body.Major,
                req.body.Address,
                req.file.filename
            ]
        );

        const [rows] = await connection.execute(
            "SELECT s.*, m.Description MajorDescription " +
            "FROM students s INNER JOIN major m on s.Major = m.MajorCode"
        );
        res.json(rows);
    }
);
